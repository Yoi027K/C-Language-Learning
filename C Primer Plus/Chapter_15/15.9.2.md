# 编写一个程序，通过命令行参数读取两个二进制字符串，对这两个二进制数使用～运算符、&运算符、|运算符和^运算符，并以二进制字符串形式打印结果（如果无法使用命令行环境，可以通过交互式让程序读取字符串）。


## 解答：


```c
#include <stdio.h>
#include <string.h>

unsigned int trans(char* s)
{
    unsigned int ans = 0;
    unsigned int flag = 1;
    int len = strlen(s);
    for(int i=len-1;i>=0;i--){
        if(s[i]=='1'){
            ans += flag;
        }
        flag *= 2;
    }
    return ans;
}

void Print(unsigned int n)
{
    int bits = 32;
    int started = 0;
    for(int i=bits-1;i>=0;i--){
        if(((n>>i) & 1) == 1){
            started = 1;
        }
        if(started){
            printf("%d",(n>>i) & 1);
        }
    }
    printf("\n");
}

int main()
{
    char s1[100];
    char s2[100];

    scanf("%s %s",s1,s2);

    int len1 = strlen(s1);
    int len2 = strlen(s2);

    unsigned int n1 = trans(s1);
    unsigned int n2 = trans(s2);

    unsigned int mask1 = (1u << len1) - 1;
    Print((~n1) & mask1);

    unsigned int mask2 = (1u << len2) - 1;
    Print((~n2) & mask2);

    Print(n1 & n2);
    Print(n1 | n2);
    Print(n1 ^ n2);
    
    return 0;
}
