# 4.编写一个程序，创建一个有两个成员的结构模板：<br><br>a.第1个成员是社会保险号，第2个成员是一个有3个成员的结构，第1个成员代表名，第2个成员代表中间名，第3个成员表示姓。创建并初始化一个内含5个该类型结构的数组。该程序以下面的格式打印数据：<br><br>Dribble, Flossie M.–– 302039823<br><br>如果有中间名，只打印它的第1个字母，后面加一个点（.）；如果没有中间名，则不用打印点。编写一个程序进行打印，把结构数组传递给这个函数。


## 解答：


```c
#include <stdio.h>
#include <string.h>

typedef struct{
    int num;
    struct{
        char last_name[20];
        char mid[20];
        char first_name[20];
    }name;
}people;

void f(people a)
{
    printf("%s,%s ",a.name.first_name,a.name.last_name);
    if(strcmp(a.name.mid,"#")){
        printf("%c.",a.name.mid[0]);
    }
    printf("--%d",a.num);
}

int main()
{
    people ps[5];
    printf("please inter your number: \n");
    scanf("%d",&ps[0].num);
    printf("please inter your first name: \n");
    scanf("%s",ps[0].name.first_name);
    printf("please inter your last name: \n");
    scanf("%s",ps[0].name.last_name);
    printf("please inter your mid name(inter # to skip): \n");
    scanf("%s",ps[0].name.mid);
    f(ps[0]);
    return 0;
}
```
